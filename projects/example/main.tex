\section{Setting}

\begin{definition}[Users and data]
Let $\mathcal{V}$ be the token vocabulary and $\X$ the set of finite token sequences.

Let $\U = \{1,\dots,N\}$ be the users. A user $u$ has a text distribution $D_u$ on $\X$ and a document tuple $X_u = (x_u^{(1)},\dots,x_u^{(n_u)})$.

Datasets, user fields, $N$ and $(n_u)_{u\in\U}$ follow Kandpal et al.~\cite{kandpal2024}: Reddit Comments, CC News, Enron Emails.
\end{definition}

\begin{assumption}[Within-user i.i.d.]\label{ass:iid}
$X_u \sim D_u^{\,n_u}$, independently across users.
\end{assumption}

\begin{definition}[Attacker texts and training texts]
For each user $u$, let $A_u \subset X_u$ be a uniformly random subset of size $m_u$

Let $F_u$ be the remaining documents, after the validation/test reservation of \cite{kandpal2024}. Under Assumption~\ref{ass:iid}, $A_u \sim D_u^{\,m_u}$ and $A_u$ is independent of $F_u$: the attacker texts are fresh samples from the user.
\end{definition}

\begin{definition}[Membership design]
Draw $Z_u \overset{\text{i.i.d.}}{\sim} \Bern(1/2)$ for $u\in\U$, independently of all data. Members: $S = \{u : Z_u = 1\}$. Non-members: $H = \U \setminus S$. Fine-tuning set: $D_{\mathrm{FT}} = \biguplus_{u\in S} F_u$.
\end{definition}

\begin{definition}[Fixed checkpoint]
$\theta = \Train(\theta_0, D_{\mathrm{FT}}; \xi)$, where $\theta_0$ is pre-trained GPT-Neo (125M or 1.3B), $\Train$ is full fine-tuning with the configuration of \cite{kandpal2024}, $\xi$ is a fixed seed.
All subsequent probabilities are conditional on $(Z, X, \xi)$; hence $\theta$, $S$, $H$ are fixed.
\end{definition}

\section{Threat model}

\begin{definition}[Access]
The attacker has black box access to $\theta = \theta_{FT}$ and to the reference $\theta_0 = \theta_{pre}$:
on input $x = (x_1,\dots,x_L)$, the models returns the next-token distributions
$p_\theta(\cdot \mid x_{<t})$ and $p_{\theta_0}(\cdot \mid x_{<t})$ for $t = 1,\dots,L$.
\end{definition}

\begin{definition}[Record-level scores]
Let $\ell_t = \log p_\theta(x_t \mid x_{<t})$ and $\log p_\theta(x) = \sum_{t=1}^{L} \ell_t$.
Fix $k \in (0,1]$ before any evaluation. Let $K_k(\cdot)$ return the indices of the
$\lceil kL \rceil$ smallest entries of a length-$L$ vector.
All scores are oriented so that larger means more member-like.
\begin{align*}
s_1(x) &= \tfrac{1}{L}\textstyle\sum_{t=1}^{L} \ell_t
  && \text{(LOSS \cite{yeom2018})}\\
s_2(x) &= \log p_\theta(x) - \log p_{\theta_0}(x)
  && \text{(Ref \cite{kandpal2024})}\\
s_3(x) &= \log p_\theta(x) \,/\, z(x)
  && \text{(zlib \cite{carlini2021})}\\
s_4(x) &= \tfrac{1}{|K_k(\ell)|}\textstyle\sum_{t\in K_k(\ell)} \ell_t
  && 
s_5(x) &= \tfrac{1}{|K_k(w)|}\textstyle\sum_{t\in K_k(w)} w_t
  && 
\end{align*}
Here $z(x)$ is the zlib-compressed length (bytes) of the text of $x$, and
$w_t = (\ell_t - \mu_t)/\sigma_t$, with $\mu_t$ and $\sigma_t^2$ the mean and variance of
$\log p_\theta(v \mid x_{<t})$ for $v \sim p_\theta(\cdot \mid x_{<t})$.
\end{definition}

\begin{definition}[User-level scores]
For a finite tuple of texts $A$ and $j \in \{1,\dots,5\}$,
\[
T_j(A) = \frac{1}{|A|} \sum_{x\in A} s_j(x).
\]
$T_2$ is the user inference statistic $\hat{T}$ of \cite{kandpal2024}:
\[
T_2(A) = \frac{1}{|A|} \sum_{x\in A} \log p_\theta(x) - \log p_{\theta_0}(x).
\]
\end{definition}

\begin{definition}[Toolbox]
The attacker defines a decision rule $\varphi$, based on set a threshold $\tau$ such that the fine decision $\F$ is defined as follows:
\[
\F = \bigl\{ \varphi_{j,\tau,\sigma}(A) = \Ind\{\sigma\, T_j(A) \ge \tau\} \;:\;
j \in \{1,\dots,5\},\ \tau \in \R,\ \sigma \in \{-1,+1\} \bigr\}.
\]
\end{definition}

\section{The game}

\begin{definition}[Game $G(\theta, S, H)$]\label{def:game}
To perform the UIA, we adopt the following approach:
\begin{enumerate}
  \item Draw $b \sim \Bern(1/2)$.
  \item If $b = 1$, draw $u \sim \Unif(S)$ (members); otherwise draw $u \sim \Unif(H)$ (non-members).
  \item Draw fresh texts $A \sim D_u^{\,m_u}$ and give $A$ to the attacker.
  \item The attacker queries $\theta$ and $\theta_0$, outputs $\hat b = \varphi(A)$.
\end{enumerate}
\end{definition}

\begin{definition}[Advantage]
For a decision rule $\varphi$ (possibly randomized),
\[
\TPR(\varphi) = \frac{1}{|S|}\sum_{u\in S} \Prob_{A\sim D_u^{m_u}}[\varphi(A)=1],
\qquad
\FPR(\varphi) = \frac{1}{|H|}\sum_{u\in H} \Prob_{A\sim D_u^{m_u}}[\varphi(A)=1],
\]
$\Adv(\varphi) = \TPR(\varphi) - \FPR(\varphi)$, and the target is
\[
\Adv^{*}_{\F}(\theta) = \sup_{\varphi\in\F} \Adv(\varphi).
\]
In the data of \cite{kandpal2024}, $A_u$ is one realization of step~3 for each user.
\end{definition}

Let $M_S = \frac{1}{|S|}\sum_{u\in S} D_u^{\,m_u}$ (members) and
$M_H = \frac{1}{|H|}\sum_{u\in H} D_u^{\,m_u}$ (non-members), laws on finite tuples of texts.

\begin{proposition}[Unrestricted optimum is model-free]\label{prop:unrestricted}
For fixed $(\theta, S, H)$, the supremum of $\Adv(\varphi)$ over all randomized rules
$\varphi$ equals $\TV(M_S, M_H)$. It does not depend on $\theta$.
\end{proposition}

\begin{proof}
Since $\theta$ and $\theta_0$ are fixed, every answer is a deterministic function of the query.
Any attacker is therefore a measurable map $\varphi : A \mapsto [0,1]$, the probability of outputting $1$.
Under $b=1$, $A \sim M_S$; under $b=0$, $A \sim M_H$. Hence
$\sup_\varphi \Adv(\varphi) = \sup_{\varphi} \bigl(\E_{M_S}\varphi - \E_{M_H}\varphi\bigr) = \TV(M_S, M_H)$.
\end{proof}

In the user inference game, hypotheses focus on membership, not model parameters. Fixed models $\theta$ and $\theta_0$ serve as measurement tools, yielding scores like $\log p_\theta(x) - \log p_{\theta_0}(x)$ for each user. The attack separates two histograms of these scores: one for members and one for non-members. This separation's advantage is bounded by $\TV(M_S, M_H)$, with the best threshold rule achieving the Kolmogorov Smirnov (KS) distance between them. Separating texts generated by $\theta$ from those by $\theta_0$ represents a different game, measuring fine-tuning rather than membership, with an optimal advantage of $\TV(P_\theta, P_{\theta_0})$.

So the 2 games are not equivalent: the KS distance is not necessarily equal to the TV distance. The former measures how well we can distinguish between two distributions, while the latter measures the total variation between them.

\begin{remark}
$Z_u$ is the design coin, drawn before training. $b$ is the referee's coin, drawn after training.
$\Adv = \TPR - \FPR$ does not depend on $\Prob(b=1)$; the balanced choice is a convention.
\end{remark}

\begin{remark}[Deviations from \cite{kandpal2024}]\leavevmode
\begin{enumerate}[label=(\roman*)]
  \item \emph{Membership.} \cite{kandpal2024} split users into two fixed halves.
        Here $Z_u \sim \Bern(1/2)$ independently, so $|S|$ is random with mean $N/2$.
        This randomization is required for identification (Step~2).
  \item \emph{Access.} $s_5$ needs full next-token distributions; \cite{kandpal2024} assume
        sequence likelihoods only. Scores $s_1$--$s_4$ remain valid under their access model.
  \item \emph{Assumption~\ref{ass:iid}.} It fails under within-user duplicates
        (about $11$ of CC News articles have exact duplicates within their domain,
        \cite{kandpal2024}, Appendix~D.5) and under threads (Enron).
        A duplicate of a training text in $A_u$ turns user inference into membership inference.
        Deduplicate $A_u$ against $F_u$ for each user.
  \item \emph{Pre-training overlap.} The fine-tuning datasets partially overlap with The Pile.
        Since $Z$ is independent of the data, this affects members and non-members alike:
        it weakens the signal but does not bias the member/non-member comparison.
\end{enumerate}
\end{remark}

\begin{remark}
The game covers one checkpoint, a population average over users, and rules in $\F$ only.
It does not cover retraining randomness, per-user guarantees, or attacks outside $\F$.
Estimation of $\Adv^{*}_{\F}(\theta)$ from one draw of $A_u$ per user is deferred to Step~3;
note that users supply independent but non-identically distributed scores.
\end{remark}
